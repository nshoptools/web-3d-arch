cmake_minimum_required(VERSION 3.24)
project(arch_curved_products_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ARCH_CURVED_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ARCH_MECHANICS_FIXTURES ON CACHE BOOL "" FORCE)
add_subdirectory("${ARCH_CURVED_SOURCE_ROOT}/src/kernel/mechanics" src/kernel/mechanics)
set(ARCH_SOURCE_FIXTURES ON CACHE BOOL "" FORCE)
add_subdirectory("${ARCH_CURVED_SOURCE_ROOT}/src/kernel/source-assembly" src/kernel/source-assembly)
add_executable(curved_fixture fixture.cpp "${ARCH_CURVED_SOURCE_ROOT}/src/kernel/native/product-bridge.cpp")
target_include_directories(curved_fixture PRIVATE "${ARCH_CURVED_SOURCE_ROOT}/src/kernel/native")
target_link_libraries(curved_fixture PRIVATE arch_source_assembly)
if(EMSCRIPTEN)
  target_compile_options(curved_fixture PRIVATE -fexceptions)
  set_target_properties(curved_fixture PROPERTIES SUFFIX ".js")
  target_link_options(curved_fixture PRIVATE -fexceptions -sNODERAWFS=1 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304)
endif()
