cmake_minimum_required(VERSION 3.24)
project(arch_derived_guard_regressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ARCH_GUARD_MANIFOLD_LIBRARY AND ARCH_GUARD_CLIPPER_LIBRARY)
  add_library(manifold STATIC IMPORTED GLOBAL)
  set_target_properties(manifold PROPERTIES IMPORTED_LOCATION "${ARCH_GUARD_MANIFOLD_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${ARCH_MANIFOLD_SOURCE}/include;${ARCH_GUARD_MANIFOLD_GENERATED}")
  add_library(Clipper2::Clipper2 STATIC IMPORTED GLOBAL)
  set_target_properties(Clipper2::Clipper2 PROPERTIES IMPORTED_LOCATION "${ARCH_GUARD_CLIPPER_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${ARCH_GUARD_CLIPPER_INCLUDE}")
endif()
if(NOT ARCH_GUARD_MECHANICS_SOURCE)
  get_filename_component(ARCH_GUARD_MECHANICS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../src/kernel/mechanics" ABSOLUTE)
endif()
add_subdirectory("${ARCH_GUARD_MECHANICS_SOURCE}" mechanics)
add_executable(derived_guard_probe probe.cpp)
target_link_libraries(derived_guard_probe PRIVATE arch_mechanics)
if(MSVC)
  target_compile_options(derived_guard_probe PRIVATE /fp:strict)
else()
  target_compile_options(derived_guard_probe PRIVATE -fno-fast-math -ffp-contract=off)
endif()
if(EMSCRIPTEN)
  set_target_properties(derived_guard_probe PROPERTIES SUFFIX ".js")
  target_compile_options(derived_guard_probe PRIVATE -fexceptions)
  target_link_options(derived_guard_probe PRIVATE -fexceptions -sNODERAWFS=1 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304)
endif()

add_executable(closure_probe closure.cpp)
target_include_directories(closure_probe PRIVATE "${ARCH_GUARD_MECHANICS_SOURCE}/src")
target_link_libraries(closure_probe PRIVATE manifold Clipper2::Clipper2)
if(MSVC)
  target_compile_options(closure_probe PRIVATE /fp:strict)
else()
  target_compile_options(closure_probe PRIVATE -fno-fast-math -ffp-contract=off)

if(EMSCRIPTEN)

  target_compile_options(closure_probe PRIVATE -fexceptions)
  target_link_options(closure_probe PRIVATE -fexceptions -sNODERAWFS=1 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304)
endif()
