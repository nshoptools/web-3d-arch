cmake_minimum_required(VERSION 3.24)
project(arch_mechanics VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_JSBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PYBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION ON CACHE BOOL "" FORCE)
set(MANIFOLD_USE_BUILTIN_CLIPPER2 ON CACHE BOOL "" FORCE)
set(MANIFOLD_DOWNLOADS OFF CACHE BOOL "" FORCE)
if(NOT TARGET manifold)
  if(NOT ARCH_CLIPPER2_SOURCE OR NOT EXISTS "${ARCH_CLIPPER2_SOURCE}/CPP/CMakeLists.txt")
    message(FATAL_ERROR "Select the pinned read-only Clipper2 derived source")
  endif()
  set(FETCHCONTENT_SOURCE_DIR_CLIPPER2 "${ARCH_CLIPPER2_SOURCE}" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_Clipper2 "${ARCH_CLIPPER2_SOURCE}")
endif()
if(NOT ARCH_MANIFOLD_SOURCE AND NOT TARGET manifold)
  message(FATAL_ERROR "Select pinned Manifold source explicitly")
endif()
if(EMSCRIPTEN)
  add_compile_options(-fexceptions)
  add_link_options(-fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304)
endif()
if(NOT TARGET manifold)
  add_subdirectory("${ARCH_MANIFOLD_SOURCE}" manifold EXCLUDE_FROM_ALL)
endif()
add_library(arch_mechanics STATIC src/mechanics.cpp)
target_include_directories(arch_mechanics PUBLIC src)
# Finite checks and outward-rounded interval predicates require IEEE semantics.
if(MSVC)
  target_compile_options(arch_mechanics PRIVATE /fp:strict)
else()
  target_compile_options(arch_mechanics PRIVATE -fno-fast-math -ffp-contract=off)
endif()
target_link_libraries(arch_mechanics PUBLIC manifold Clipper2::Clipper2)
option(ARCH_MECHANICS_FIXTURES "Build synthetic fixture executable (test I/O only)" OFF)
if(ARCH_MECHANICS_FIXTURES)
  add_executable(mechanics_fixture tests/fixture.cpp)
  target_link_libraries(mechanics_fixture PRIVATE arch_mechanics)
  if(EMSCRIPTEN)
    set_target_properties(mechanics_fixture PROPERTIES SUFFIX ".js")
    target_link_options(mechanics_fixture PRIVATE -sNODERAWFS=1 -sENVIRONMENT=node)
  endif()
endif()
if(ARCH_SOURCE_ASSEMBLY_TEST_SOURCE)
  # Optional integration regression against a separately checked source target.
  # Its input tree stays read-only; all objects/binaries stay in this build.
  set(ARCH_SOURCE_FIXTURES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ARCH_SOURCE_ASSEMBLY_TEST_SOURCE}" source-integration EXCLUDE_FROM_ALL)
  add_executable(source_mechanics_fixture "${ARCH_SOURCE_ASSEMBLY_TEST_SOURCE}/tests/fixture.cpp")
  target_link_libraries(source_mechanics_fixture PRIVATE arch_source_assembly)
  if(EMSCRIPTEN)
    set_target_properties(source_mechanics_fixture PROPERTIES SUFFIX ".js")
    target_link_options(source_mechanics_fixture PRIVATE -sNODERAWFS=1 -sENVIRONMENT=node)
  endif()
endif()
