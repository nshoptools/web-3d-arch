cmake_minimum_required(VERSION 3.24)
project(arch_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_JSBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION ON CACHE BOOL "" FORCE)
set(MANIFOLD_USE_BUILTIN_CLIPPER2 ON CACHE BOOL "" FORCE)
set(MANIFOLD_DOWNLOADS OFF CACHE BOOL "" FORCE)
if(NOT ARCH_CLIPPER2_SOURCE OR NOT EXISTS "${ARCH_CLIPPER2_SOURCE}/CPP/CMakeLists.txt")
  message(FATAL_ERROR "Prepare the pinned Clipper2 original/derived sources first")
endif()
set(FETCHCONTENT_SOURCE_DIR_CLIPPER2 "${ARCH_CLIPPER2_SOURCE}" CACHE PATH "" FORCE)
# Manifold's no-download diagnostic uses the original dependency spelling.
set(FETCHCONTENT_SOURCE_DIR_Clipper2 "${ARCH_CLIPPER2_SOURCE}")
if(NOT ARCH_MANIFOLD_SOURCE)
  message(FATAL_ERROR "ARCH_MANIFOLD_SOURCE must select the pinned dependency")
endif()
add_subdirectory("${ARCH_MANIFOLD_SOURCE}" manifold)
if(NOT ARCH_HARFBUZZ_SOURCE OR NOT EXISTS "${ARCH_HARFBUZZ_SOURCE}/src/hb.h")
  message(FATAL_ERROR "ARCH_HARFBUZZ_SOURCE must select the HarfBuzz revision in toolchain-lock.json")
endif()
foreach(hb_option HB_HAVE_CAIRO HB_HAVE_FREETYPE HB_HAVE_GRAPHITE2 HB_HAVE_GLIB HB_HAVE_ICU HB_HAVE_UNISCRIBE HB_HAVE_GDI HB_HAVE_DIRECTWRITE HB_BUILD_UTILS HB_BUILD_SUBSET HB_BUILD_RASTER HB_BUILD_VECTOR HB_BUILD_GPU HB_HAVE_GOBJECT HB_HAVE_INTROSPECTION)
  set(${hb_option} OFF CACHE BOOL "" FORCE)
endforeach()
set(HB_BUILD_GPU_DEMO OFF CACHE STRING "" FORCE)
add_subdirectory("${ARCH_HARFBUZZ_SOURCE}" harfbuzz EXCLUDE_FROM_ALL)
add_library(arch_geometry STATIC geometry.cpp)
target_sources(arch_geometry PRIVATE conditioning.cpp conditioning-api.cpp)
if(MSVC)
  target_compile_options(arch_geometry PRIVATE /fp:strict)
elseif(EMSCRIPTEN)
  target_compile_options(arch_geometry PRIVATE -ffp-contract=off)
endif()
target_include_directories(arch_geometry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(arch_geometry PUBLIC manifold Clipper2::Clipper2 harfbuzz)
if(EMSCRIPTEN)
  target_compile_options(arch_geometry PRIVATE -fexceptions)
endif()
option(ARCH_NATIVE_SMOKE "Build the analytic fixture harness" OFF)
option(ARCH_FONT_PROBE "Build the test-only native HarfBuzz parity probe" OFF)
if(ARCH_FONT_PROBE)
  add_executable(arch_font_probe "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/kernel/harfbuzz-native.cpp")
  target_link_libraries(arch_font_probe PRIVATE harfbuzz)
  add_executable(arch_geometry_performance "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/kernel/geometry-performance.cpp")
  target_link_libraries(arch_geometry_performance PRIVATE arch_geometry)
endif()
if(ARCH_NATIVE_SMOKE)
  add_executable(arch_geometry_smoke smoke.cpp)
  target_link_libraries(arch_geometry_smoke PRIVATE arch_geometry)
endif()
install(TARGETS arch_geometry ARCHIVE DESTINATION lib)

# Both libraries share the pinned Manifold/Clipper allocator and target flags.
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../mechanics" mechanics)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../source-assembly" source-assembly)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../final-scene-export" final-scene-export)
if(ARCH_PRINTING_SOURCE)
  set(ARCH_PRINTING_STANDALONE OFF CACHE BOOL "" FORCE)
  set(ARCH_PRINTING_PTHREADS ON CACHE BOOL "" FORCE)
  set(ARCH_KERNEL_RUNTIME_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_subdirectory("${ARCH_PRINTING_SOURCE}" printing)
endif()

# Product runtime: source/mesh bridge in the existing geometry archive.
target_sources(arch_geometry PRIVATE product-bridge.cpp)
target_include_directories(arch_geometry PRIVATE "../source-assembly/src" "../mechanics/src")

# Imported STL/OBJ and general CSG share this Manifold and this allocator.
set(ARCHMI_MECHANICS_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../mechanics/src")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../mesh-import" mesh-import)
