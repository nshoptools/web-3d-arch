cmake_minimum_required(VERSION 3.24)
project(arch_source_assembly VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
  add_compile_options(-fexceptions)
  add_link_options(-fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304)
endif()
if(NOT TARGET arch_mechanics)
  if(NOT ARCH_MECHANICS_SOURCE)
    message(FATAL_ERROR "Provide mechanics ABI2 source, including the documented source-datum extension for layer integration")
  endif()
  set(ARCH_MECHANICS_FIXTURES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ARCH_MECHANICS_SOURCE}" mechanics EXCLUDE_FROM_ALL)
endif()
add_library(arch_source_assembly STATIC src/source_assembly.cpp)
target_include_directories(arch_source_assembly PUBLIC src)
# Finite checks and outward-rounded interval predicates require IEEE semantics.
if(MSVC)
  target_compile_options(arch_source_assembly PRIVATE /fp:strict)
else()
  target_compile_options(arch_source_assembly PRIVATE -fno-fast-math -ffp-contract=off)
endif()
target_link_libraries(arch_source_assembly PUBLIC arch_mechanics manifold Clipper2::Clipper2)
option(ARCH_SOURCE_FIXTURES "Build prepared source fixtures" OFF)
if(ARCH_SOURCE_FIXTURES)
  add_executable(source_fixture tests/fixture.cpp)
  target_link_libraries(source_fixture PRIVATE arch_source_assembly)
  add_executable(mechanics_bridge_fixture tests/mechanics_fixture.cpp)
  target_link_libraries(mechanics_bridge_fixture PRIVATE arch_mechanics)
  if(EMSCRIPTEN)
    set_target_properties(source_fixture PROPERTIES SUFFIX ".js")
    target_link_options(source_fixture PRIVATE -sNODERAWFS=1 -sENVIRONMENT=node)
    set_target_properties(mechanics_bridge_fixture PROPERTIES SUFFIX ".js")
    target_link_options(mechanics_bridge_fixture PRIVATE -sNODERAWFS=1 -sENVIRONMENT=node)
  endif()
endif()
