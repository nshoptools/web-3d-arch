cmake_minimum_required(VERSION 3.24)
project(arch_source_domain_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SOURCE_CHILD "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT ARCH_ROOT_NATIVE_SOURCE)
 message(FATAL_ERROR "Provide read-only root native bridge directory")
endif()
set(ARCH_SOURCE_FIXTURES OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE_CHILD}" child EXCLUDE_FROM_ALL)
foreach(label before candidate)
 if(label STREQUAL "before")
  set(source "${CMAKE_CURRENT_LIST_DIR}/fixtures/source_assembly.before.cpp")
 else()
  set(source "${SOURCE_CHILD}/src/source_assembly.cpp")
 endif()
 # Test-only observation: identical implementation except capture of the input
 # to regularize_support. No predicate, epsilon, operation or assertion changes.
 file(READ "${source}" body)
 string(REPLACE "  CrossSection regularize_support(const CrossSection&s){" "  CrossSection regularize_support(const CrossSection&s){ arch_test_support(s.ToPolygons());" body "${body}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${label}-source.cpp" "#include <manifold/cross_section.h>\nvoid arch_test_support(const manifold::Polygons&);\n${body}")
 add_library(${label}_source STATIC "${CMAKE_CURRENT_BINARY_DIR}/${label}-source.cpp")
 target_include_directories(${label}_source PUBLIC "${SOURCE_CHILD}/src")
 target_link_libraries(${label}_source PUBLIC arch_mechanics manifold Clipper2::Clipper2)
 add_executable(replay_${label} replay.cpp "${ARCH_ROOT_NATIVE_SOURCE}/product-bridge.cpp")
 target_include_directories(replay_${label} PRIVATE "${ARCH_ROOT_NATIVE_SOURCE}")
 target_link_libraries(replay_${label} PRIVATE ${label}_source)
 if(MSVC)
  target_compile_options(${label}_source PRIVATE /fp:strict)
 else()
  target_compile_options(${label}_source PRIVATE -fno-fast-math -ffp-contract=off)
 endif()
 if(EMSCRIPTEN)
  target_compile_options(${label}_source PRIVATE -fexceptions)
  target_compile_options(replay_${label} PRIVATE -fexceptions)
  set_target_properties(replay_${label} PROPERTIES SUFFIX ".js")
  target_link_options(replay_${label} PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sNODERAWFS=1 -sENVIRONMENT=node)
 endif()
endforeach()
add_executable(support_paths support_paths.cpp)
target_include_directories(support_paths PRIVATE "${SOURCE_CHILD}/src")
target_link_libraries(support_paths PRIVATE Clipper2::Clipper2)
if(MSVC)
 target_compile_options(support_paths PRIVATE /fp:strict)
else()
 target_compile_options(support_paths PRIVATE -fno-fast-math -ffp-contract=off)
endif()
if(EMSCRIPTEN)
 target_compile_options(support_paths PRIVATE -fexceptions)
 set_target_properties(support_paths PROPERTIES SUFFIX ".js")
 target_link_options(support_paths PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sNODERAWFS=1 -sENVIRONMENT=node)
endif()
# The same production target, without the observation hook, must also build.
add_executable(replay_production replay.cpp "${ARCH_ROOT_NATIVE_SOURCE}/product-bridge.cpp")
target_include_directories(replay_production PRIVATE "${ARCH_ROOT_NATIVE_SOURCE}")
target_link_libraries(replay_production PRIVATE arch_source_assembly)
if(EMSCRIPTEN)
 target_compile_options(replay_production PRIVATE -fexceptions)
 set_target_properties(replay_production PROPERTIES SUFFIX ".js")
 target_link_options(replay_production PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sNODERAWFS=1 -sENVIRONMENT=node)
endif()
