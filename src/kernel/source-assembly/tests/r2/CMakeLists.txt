cmake_minimum_required(VERSION 3.22)
project(remediation_r2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(EMSCRIPTEN)
  add_compile_options(-pthread -fexceptions)
  add_link_options(-pthread -fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sNODERAWFS=1 -sENVIRONMENT=node)
endif()
set(ARCH_MECHANICS_FIXTURES OFF CACHE BOOL "" FORCE)
set(ARCH_SOURCE_FIXTURES OFF CACHE BOOL "" FORCE)
add_subdirectory("${CANDIDATE}/src/kernel/mechanics" mechanics)
add_subdirectory("${CANDIDATE}/src/kernel/source-assembly" source)
add_library(arch_mechanics_preimage STATIC "${PREIMAGE}/src/kernel/mechanics/src/mechanics.cpp")
target_include_directories(arch_mechanics_preimage PUBLIC "${PREIMAGE}/src/kernel/mechanics/src")
target_link_libraries(arch_mechanics_preimage PUBLIC manifold Clipper2::Clipper2)
add_library(arch_source_preimage STATIC "${PREIMAGE}/src/kernel/source-assembly/src/source_assembly.cpp")
target_include_directories(arch_source_preimage PUBLIC "${PREIMAGE}/src/kernel/source-assembly/src")
target_link_libraries(arch_source_preimage PUBLIC arch_mechanics_preimage manifold Clipper2::Clipper2)
foreach(phase IN ITEMS old new)
  if(phase STREQUAL old)
    set(tree "${PREIMAGE}")
    set(mech arch_mechanics_preimage)
    set(src arch_source_preimage)
  else()
    set(tree "${CANDIDATE}")
    set(mech arch_mechanics)
    set(src arch_source_assembly)
  endif()
  add_executable(mechanics_${phase} "${tree}/src/kernel/mechanics/tests/fixture.cpp")
  target_link_libraries(mechanics_${phase} PRIVATE "${mech}")
  add_executable(source_${phase} "${tree}/src/kernel/source-assembly/tests/fixture.cpp")
  target_link_libraries(source_${phase} PRIVATE "${src}")
  if(EMSCRIPTEN)
    set_target_properties(mechanics_${phase} source_${phase} PROPERTIES SUFFIX ".js")
  endif()
endforeach()

add_executable(source_probe_old "${CANDIDATE}/src/kernel/source-assembly/tests/fixture.cpp")
target_compile_definitions(source_probe_old PRIVATE ARCH_R2_PREIMAGE=1)
target_link_libraries(source_probe_old PRIVATE arch_source_preimage)
if(EMSCRIPTEN)
  set_target_properties(source_probe_old PROPERTIES SUFFIX ".js")
endif()

add_executable(mechanics_probe_old "${CANDIDATE}/src/kernel/mechanics/tests/fixture.cpp")
target_compile_definitions(mechanics_probe_old PRIVATE ARCH_R2_PREIMAGE=1)
target_link_libraries(mechanics_probe_old PRIVATE arch_mechanics_preimage)
if(EMSCRIPTEN)
  set_target_properties(mechanics_probe_old PROPERTIES SUFFIX ".js")
endif()
