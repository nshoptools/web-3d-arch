cmake_minimum_required(VERSION 3.24)
if(NOT TARGET manifold OR NOT TARGET Clipper2)
  message(FATAL_ERROR "Add this child after the existing pinned Manifold/Clipper2 targets")
endif()
add_library(arch_final_scene_export STATIC native/final_scene_export.cpp)
target_compile_features(arch_final_scene_export PRIVATE cxx_std_17)
target_include_directories(arch_final_scene_export PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native" "${CMAKE_CURRENT_SOURCE_DIR}/../native")
target_link_libraries(arch_final_scene_export PUBLIC manifold Clipper2::Clipper2)
if(MSVC)
  target_compile_options(arch_final_scene_export PRIVATE /fp:strict)
else()
  target_compile_options(arch_final_scene_export PRIVATE -ffp-contract=off)
endif()
if(EMSCRIPTEN)
  target_compile_options(arch_final_scene_export PRIVATE -fexceptions)
endif()
option(ARCH_FINAL_EXPORT_TESTS "Build synthetic final-CSG fixture entry points" OFF)
if(ARCH_FINAL_EXPORT_TESTS)
  target_sources(arch_final_scene_export PRIVATE tests/fixture.cpp)
endif()
