cmake_minimum_required(VERSION 3.24)
project(arch_printing VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ARCH_PRINTING_STANDALONE "Build a separate demo module/DLL (not used by unified kernel)" ${PROJECT_IS_TOP_LEVEL})
option(ARCH_PRINTING_PTHREADS "Compile Emscripten objects for the shared kernel heap" ON)
if(NOT ARCH_LIB3MF_SOURCE OR NOT EXISTS "${ARCH_LIB3MF_SOURCE}/.arch-printing-prepared.json")
  message(FATAL_ERROR "Prepare pinned sources with tools/prepare-lib3mf.py; configure never downloads")
endif()
set(LIB3MF_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LIB3MF_BUILD_WASM OFF CACHE BOOL "" FORCE)
set(LIB3MF_TESTS OFF CACHE BOOL "" FORCE)
set(USE_INCLUDED_ZLIB ON CACHE BOOL "" FORCE)
set(USE_INCLUDED_LIBZIP ON CACHE BOOL "" FORCE)
set(USE_INCLUDED_CPPBASE64 ON CACHE BOOL "" FORCE)
set(USE_INCLUDED_FASTFLOAT ON CACHE BOOL "" FORCE)
set(USE_PLATFORM_UUID OFF CACHE BOOL "" FORCE)
set(buildcache_program "" CACHE FILEPATH "No shared buildcache launcher" FORCE)
add_subdirectory("${ARCH_LIB3MF_SOURCE}" lib3mf EXCLUDE_FROM_ALL)
add_library(arch3mf_static STATIC src/native/arch3mf.cpp)
add_library(arch_printing::core ALIAS arch3mf_static)
target_include_directories(arch3mf_static PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/native")
target_compile_definitions(arch3mf_static PUBLIC ARCH3MF_STATIC=1)
target_link_libraries(arch3mf_static PUBLIC lib3mf)
if(ARCH_KERNEL_RUNTIME_INCLUDE)
  if(NOT EXISTS "${ARCH_KERNEL_RUNTIME_INCLUDE}/runtime.h")
    message(FATAL_ERROR "ARCH_KERNEL_RUNTIME_INCLUDE must contain runtime ABI 2 header")
  endif()
  target_sources(arch3mf_static PRIVATE src/native/kernel_bridge.cpp)
  target_include_directories(arch3mf_static PRIVATE "${ARCH_KERNEL_RUNTIME_INCLUDE}")
endif()
if(EMSCRIPTEN)
  if(NOT ARCH_PRINTING_PTHREADS)
    message(FATAL_ERROR "Unified printing requires -pthread")
  endif()
  foreach(t lib3mf arch3mf_static)
    target_compile_options(${t} PRIVATE -pthread "$<$<COMPILE_LANGUAGE:CXX>:-fexceptions>")
  endforeach()
  target_link_options(arch3mf_static INTERFACE -pthread -fexceptions)
endif()
set_target_properties(arch3mf_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ARCH_PRINTING_STANDALONE)
  if(EMSCRIPTEN)
    add_executable(arch3mf src/native/wasm_entry.cpp)
    target_link_libraries(arch3mf PRIVATE arch3mf_static)
    set_target_properties(arch3mf PROPERTIES SUFFIX ".mjs")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cmake/core-exports.json" arch_exports)
    target_link_options(arch3mf PRIVATE --no-entry -O2 -pthread -fexceptions
      -sMODULARIZE=1 -sEXPORT_ES6=1 -sALLOW_MEMORY_GROWTH=1
      -sINITIAL_MEMORY=67108864 -sMAXIMUM_MEMORY=1073741824 -sSTACK_SIZE=4194304
      "-sEXPORTED_FUNCTIONS=${arch_exports}" "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','UTF8ToString']")
  else()
    add_library(arch3mf SHARED src/native/arch3mf.cpp)
    target_link_libraries(arch3mf PRIVATE lib3mf)
  endif()
endif()
