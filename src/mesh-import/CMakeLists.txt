cmake_minimum_required(VERSION 3.24)
project(arch_mesh_import VERSION 0.2.0 LANGUAGES CXX)
if(NOT TARGET manifold)
 message(FATAL_ERROR "Add pinned same-module Manifold3.5.3 first; no download")
endif()
get_target_property(_archmi_manifold_type manifold TYPE)
if(NOT _archmi_manifold_type STREQUAL "STATIC_LIBRARY")
 message(FATAL_ERROR "Manifold must be the same-module static target")
endif()
include(cmake/manifold-pins.cmake)
option(ARCHMI_ENABLE_3MF_EXTENSION "Experimental 3MF importer (not v1 default)" OFF)
set(_archmi_sources src/native/mesh_import.cpp)
if(ARCHMI_ENABLE_3MF_EXTENSION)
 if(NOT TARGET lib3mf OR NOT ARCH_LIB3MF_SOURCE OR NOT ARCHMI_LIBZIP_CONFIG)
  message(FATAL_ERROR "Pinned lib3mf target/source/libzip headers required for optional3MF")
 endif()
 get_target_property(_archmi_type lib3mf TYPE)
 if(NOT _archmi_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "lib3mf must be the same-module static target")
 endif()
 include(cmake/expat.cmake)
 list(APPEND _archmi_sources src/native/package_guard.cpp src/native/decode_3mf.cpp)
else()
 list(APPEND _archmi_sources src/native/decode_3mf_deferred.cpp)
endif()
add_library(arch_mesh_import_static STATIC ${_archmi_sources})
add_library(arch_mesh_import::core ALIAS arch_mesh_import_static)
target_compile_features(arch_mesh_import_static PUBLIC cxx_std_17)
target_include_directories(arch_mesh_import_static PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/native")
target_link_libraries(arch_mesh_import_static PUBLIC manifold)
if(ARCHMI_ENABLE_3MF_EXTENSION)
 target_include_directories(arch_mesh_import_static PRIVATE
  "${ARCH_LIB3MF_SOURCE}/Autogenerated/Bindings/Cpp" "${ARCH_LIB3MF_SOURCE}/Include"
  "${ARCH_LIB3MF_SOURCE}/Libraries/libzip/Include" "${ARCHMI_LIBZIP_CONFIG}")
 target_link_libraries(arch_mesh_import_static PUBLIC lib3mf expat)
endif()
add_library(arch_imported_csg_static STATIC src/native/imported_csg.cpp)
add_library(arch_imported_csg::core ALIAS arch_imported_csg_static)
target_include_directories(arch_imported_csg_static PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/native")
target_compile_features(arch_imported_csg_static PUBLIC cxx_std_17)
target_link_libraries(arch_imported_csg_static PUBLIC arch_mesh_import_static manifold)
foreach(_archmi_target arch_mesh_import_static arch_imported_csg_static)
 if(MSVC)
  target_compile_options(${_archmi_target} PRIVATE /EHsc /fp:strict)
 else()
  target_compile_options(${_archmi_target} PRIVATE -fno-fast-math -ffp-contract=off)
 endif()
 if(EMSCRIPTEN)
  target_compile_options(${_archmi_target} PRIVATE -pthread -fexceptions)
  target_link_options(${_archmi_target} INTERFACE -pthread -fexceptions)
 endif()
endforeach()
if(EMSCRIPTEN)
 target_compile_options(arch_mesh_import_static PRIVATE -pthread -fexceptions)
 target_link_options(arch_mesh_import_static INTERFACE -pthread -fexceptions)
endif()
if(MSVC)
 target_compile_options(arch_mesh_import_static PRIVATE /EHsc)
 target_link_libraries(arch_mesh_import_static PUBLIC advapi32 ole32 uuid)
endif()

if(ARCHMI_MECHANICS_INCLUDE)
 add_library(arch_mesh_root_static STATIC src/native/root_gate.cpp)
 target_compile_features(arch_mesh_root_static PUBLIC cxx_std_17)
 target_include_directories(arch_mesh_root_static PRIVATE "${ARCHMI_MECHANICS_INCLUDE}")
 target_link_libraries(arch_mesh_root_static PUBLIC arch_imported_csg_static arch_mechanics)
 if(MSVC)
  target_compile_options(arch_mesh_root_static PRIVATE /EHsc /fp:strict)
 else()
  target_compile_options(arch_mesh_root_static PRIVATE -fno-fast-math -ffp-contract=off)
 endif()
 if(EMSCRIPTEN)
  target_compile_options(arch_mesh_root_static PRIVATE -pthread -fexceptions)
  target_link_options(arch_mesh_root_static INTERFACE -pthread -fexceptions)
 endif()
endif()
